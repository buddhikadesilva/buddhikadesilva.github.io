How to Load Table Data Rows and Refresh the UI in Flutter

Originally published on Medium

In Flutter, you can dynamically load rows of tabular data and trigger reactive UI rebuilds using ListView.builder and setState. Here is a straightforward walkthrough of how to build and update dynamic rows:

Step-by-Step Implementation

1. Define Your Table Row List

Create a collection of TableRow items or model objects fetched from an API or database:

List<TableRow> tableDataRows = [
  TableRow(
    children: [
      TableCell(child: Padding(padding: EdgeInsets.all(8), child: Text('Row 1, Column 1'))),
      TableCell(child: Padding(padding: EdgeInsets.all(8), child: Text('Row 1, Column 2'))),
    ],
  ),
  TableRow(
    children: [
      TableCell(child: Padding(padding: EdgeInsets.all(8), child: Text('Row 2, Column 1'))),
      TableCell(child: Padding(padding: EdgeInsets.all(8), child: Text('Row 2, Column 2'))),
    ],
  ),
];

2. Render with ListView.builder

Use ListView.builder to lazily instantiate and display rows:

ListView.builder(
  itemCount: tableDataRows.length,
  itemBuilder: (BuildContext context, int index) {
    return Table(children: [tableDataRows[index]]);
  },
)

3. Refresh Data with setState()

When updated data arrives, update the list within a setState() callback:

setState(() {
  tableDataRows = [
    TableRow(children: [
      TableCell(child: Padding(padding: EdgeInsets.all(8), child: Text('Row 1, Updated'))),
      TableCell(child: Padding(padding: EdgeInsets.all(8), child: Text('Row 1, Value'))),
    ]),
  ];
});

Complete Working Example

import 'package:flutter/material.dart';

void main() => runApp(MyApp());

class MyApp extends StatefulWidget {
  @override
  _MyAppState createState() => _MyAppState();
}

class _MyAppState extends State<MyApp> {
  List<TableRow> tableDataRows = [
    TableRow(children: [
      TableCell(child: Text('Item A')),
      TableCell(child: Text('Qty: 10')),
    ]),
    TableRow(children: [
      TableCell(child: Text('Item B')),
      TableCell(child: Text('Qty: 25')),
    ]),
  ];

  void _refreshData() {
    setState(() {
      tableDataRows = [
        TableRow(children: [
          TableCell(child: Text('Item A (Refreshed)')),
          TableCell(child: Text('Qty: 15')),
        ]),
        TableRow(children: [
          TableCell(child: Text('Item B (Refreshed)')),
          TableCell(child: Text('Qty: 40')),
        ]),
      ];
    });
  }

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: Scaffold(
        appBar: AppBar(title: Text('Table Data Rows Example')),
        body: ListView.builder(
          padding: EdgeInsets.all(16),
          itemCount: tableDataRows.length,
          itemBuilder: (context, index) => Table(
            border: TableBorder.all(color: Colors.grey.shade300),
            children: [tableDataRows[index]],
          ),
        ),
        floatingActionButton: FloatingActionButton(
          onPressed: _refreshData,
          child: Icon(Icons.refresh),
        ),
      ),
    );
  }
}
